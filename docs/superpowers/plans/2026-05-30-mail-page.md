# Mail Page + Unread Badge Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an EVE Mail page (3-pane read/organize/compose) to the navbar plus a navbar mail icon with a red Inbox-unread badge.

**Architecture:** Thin async ESI wrappers in `esi.py`; `/api/me/mail/*` FastAPI routes in `main.py` that resolve the session → token → ESI, enriching sender/recipient names server-side. Frontend is a new `#view-mail` 3-pane grid plus a navbar icon+badge poller, all in the single `static/index.html`. No new DB tables; mail bodies cached in-memory.

**Tech Stack:** FastAPI, httpx, plain HTML/CSS/JS, pytest + respx.

**Spec:** `docs/superpowers/specs/2026-05-30-mail-page-design.md`

**Conventions (verified in this codebase):**
- ESI base: `ESI_BASE = "https://esi.evetech.net/latest"` (in `esi.py`). All mail paths end with a trailing slash, e.g. `{ESI_BASE}/characters/{cid}/mail/`.
- ESI wrappers take `(character_id, access_token, ...)`, use `async with httpx.AsyncClient()`, set `headers={"Authorization": f"Bearer {access_token}"}`, call `r.raise_for_status()`.
- Routes use `session: str | None = Cookie(None)` → `character_id = await get_current_character_id(session)`, then `try: access_token = await get_valid_token(character_id); ...` with `except httpx.HTTPStatusError` (403 → 403 with hint, else 502) and `except httpx.HTTPError` → 502.
- Existing helper `resolve_names(ids: list[int]) -> dict[int,str]` (already imported into `main.py`) resolves char/corp/alliance IDs → names via `/universe/names/`.
- Tests: `pytest` with `respx` to mock ESI; `client` fixture + session cookie via `make_session_cookie(cid)` and `upsert_token(...)` (see `tests/test_member_pvp_income.py` and `tests/test_api.py`).

**Test command:** `venv/bin/python -m pytest <path> -v`
**Run app manually:** `set -a && source .env && set +a && /home/paolo/.local/bin/uvicorn main:app --port 8000`

```python
                           labels: list[int] | None = None,
                           last_mail_id: int | None = None) -> list[dict]:
    """Up to 50 mail headers, newest first. `last_mail_id` pages older."""

            params=params,
```

```python
# ── Mail (write / organize / send) ───────────────────────────────────────────

async def set_mail(character_id: int, access_token: str, mail_id: int,
                   read: bool | None = None, labels: list[int] | None = None) -> None:
    """Mark read/unread and/or replace the mail's labels."""
    payload: dict = {}
    if read is not None:
        payload["read"] = read
    if labels is not None:
        payload["labels"] = labels
    async with httpx.AsyncClient() as client:
        r = await client.put(
            f"{ESI_BASE}/characters/{character_id}/mail/{mail_id}/",
            headers={"Authorization": f"Bearer {access_token}"},
            json=payload,
        )
        r.raise_for_status()


async def delete_mail(character_id: int, access_token: str, mail_id: int) -> None:
    async with httpx.AsyncClient() as client:
        r = await client.delete(
            f"{ESI_BASE}/characters/{character_id}/mail/{mail_id}/",
            headers={"Authorization": f"Bearer {access_token}"},
        )
        r.raise_for_status()


async def create_mail_label(character_id: int, access_token: str,
                            name: str, color: str) -> int:
    async with httpx.AsyncClient() as client:
        r = await client.post(
            f"{ESI_BASE}/characters/{character_id}/mail/labels/",
            headers={"Authorization": f"Bearer {access_token}"},
            json={"name": name, "color": color},
        )
        r.raise_for_status()
        return r.json()


async def delete_mail_label(character_id: int, access_token: str, label_id: int) -> None:
    async with httpx.AsyncClient() as client:
        r = await client.delete(
            f"{ESI_BASE}/characters/{character_id}/mail/labels/{label_id}/",
            headers={"Authorization": f"Bearer {access_token}"},
        )
        r.raise_for_status()


async def send_mail(character_id: int, access_token: str, recipients: list[dict],
                    subject: str, body: str, approved_cost: int = 0) -> int:
    async with httpx.AsyncClient() as client:
        r = await client.post(
            f"{ESI_BASE}/characters/{character_id}/mail/",
            headers={"Authorization": f"Bearer {access_token}"},
            json={"recipients": recipients, "subject": subject,
                  "body": body, "approved_cost": approved_cost},
        )
        r.raise_for_status()
        return r.json()


async def resolve_recipient_ids(names: list[str]) -> list[dict]:
    """Resolve names to mail recipients via /universe/ids/.
    Returns [{recipient_id, recipient_type, name}] for characters/corps/alliances."""
    clean = list({n for n in names if n})
    if not clean:
        return []
    type_map = {
        "characters": "character",
        "corporations": "corporation",
        "alliances": "alliance",
    }
    async with httpx.AsyncClient() as client:
        try:
            r = await client.post(f"{ESI_BASE}/universe/ids/", json=clean)
            r.raise_for_status()
            data = r.json()
        except httpx.HTTPError:
            return []
    out: list[dict] = []
    for key, rtype in type_map.items():
        for item in data.get(key, []):
            out.append({"recipient_id": item["id"],
                        "recipient_type": rtype, "name": item["name"]})
    return out
```

```python
# ── Mail ─────────────────────────────────────────────────────────────────────

# Mail bodies are immutable; cache per (character_id, mail_id) for the process.
_mail_body_cache: dict[tuple[int, int], dict] = {}

# Short cache for the Inbox-unread badge poll so 60s polling never hammers ESI.
_mail_unread_cache: dict[int, tuple[float, int]] = {}  # cid -> (cached_at, inbox_unread)
_MAIL_UNREAD_TTL = 30.0

INBOX_LABEL_ID = 1


def _mail_scope_http(e: httpx.HTTPStatusError) -> HTTPException:
    print(f"ESI mail HTTP {e.response.status_code}: {e.response.text}")
    if e.response.status_code == 403:
        return HTTPException(
            status_code=403,
            detail="Mail scope not granted — log out and back in to enable Mail.")
    return HTTPException(status_code=502, detail="ESI unavailable")


async def _enrich_mail_names(mails: list[dict]) -> None:
    """Attach `from_name` and recipient `name` to a list of mail dicts in place."""
    ids: list[int] = []
    for m in mails:
        if m.get("from"):
            ids.append(m["from"])
        for rcpt in m.get("recipients", []):
            if rcpt.get("recipient_id"):
                ids.append(rcpt["recipient_id"])
    names = await resolve_names(ids)
    for m in mails:
        m["from_name"] = names.get(m.get("from"))
        for rcpt in m.get("recipients", []):
            rcpt["name"] = names.get(rcpt.get("recipient_id"))


@app.get("/api/me/mail")
async def my_mail(labels: str | None = None, last_mail_id: int | None = None,
                  session: str | None = Cookie(None)):
    character_id = await get_current_character_id(session)
    label_ids = [int(x) for x in labels.split(",")] if labels else None
    try:
        access_token = await get_valid_token(character_id)
        headers = await get_mail_headers(character_id, access_token,
                                         labels=label_ids, last_mail_id=last_mail_id)
    except httpx.HTTPStatusError as e:
        raise _mail_scope_http(e)
    except httpx.HTTPError:
        raise HTTPException(status_code=502, detail="ESI unavailable")
    await _enrich_mail_names(headers)
    return headers


@app.get("/api/me/mail/labels")
async def my_mail_labels(session: str | None = Cookie(None)):
    character_id = await get_current_character_id(session)
    try:
        access_token = await get_valid_token(character_id)
        labels, lists = await asyncio.gather(
            get_mail_labels(character_id, access_token),
            get_mailing_lists(character_id, access_token),
        )
    except httpx.HTTPStatusError as e:
        raise _mail_scope_http(e)
    except httpx.HTTPError:
        raise HTTPException(status_code=502, detail="ESI unavailable")
    return {"labels": labels.get("labels", []),
            "total_unread_count": labels.get("total_unread_count", 0),
            "lists": lists}


@app.get("/api/me/mail/unread")
async def my_mail_unread(session: str | None = Cookie(None)):
    character_id = await get_current_character_id(session)
    hit = _mail_unread_cache.get(character_id)
    if hit and (time.time() - hit[0]) < _MAIL_UNREAD_TTL:
        return {"inbox_unread": hit[1]}
    try:
        access_token = await get_valid_token(character_id)
        labels = await get_mail_labels(character_id, access_token)
    except httpx.HTTPStatusError as e:
        raise _mail_scope_http(e)
    except httpx.HTTPError:
        raise HTTPException(status_code=502, detail="ESI unavailable")
    inbox = next((l for l in labels.get("labels", [])
```

Replace the temporary `sanitizeEveMail` with a DOM-based one that allows safe markup and neutralizes client-only links:

```javascript
  // Render EVE mail body safely: allow text, <br>, basic styling, http links;
  // neutralize client-only links (showinfo:/killReport:) to non-clickable text.
  function sanitizeEveMail(raw) {
    const out = document.createElement('div');
    // Parse the EVE markup in an inert document (no scripts/resource loads run).
    const doc = new DOMParser().parseFromString(raw, 'text/html');
    const allowed = new Set(['BR', 'B', 'I', 'U', 'FONT', 'A', 'LOC', 'SPAN', 'DIV', 'P']);

    function argbToRgb(c) {
      if (!c) return null;
      const m = String(c).replace('#', '');
      if (m.length === 8) return '#' + m.slice(2);   // strip AA from #AARRGGBB
      if (m.length === 6) return '#' + m;
      return null;
    }

    function walk(src, dst) {
      src.childNodes.forEach(node => {
        if (node.nodeType === Node.TEXT_NODE) {
          dst.appendChild(document.createTextNode(node.nodeValue));
          return;
        }
        if (node.nodeType !== Node.ELEMENT_NODE) return;
        const tag = node.tagName;
        if (!allowed.has(tag)) { walk(node, dst); return; }  // unwrap unknown tags

        if (tag === 'BR') { dst.appendChild(document.createElement('br')); return; }

        if (tag === 'A') {
          const href = node.getAttribute('href') || '';
          if (/^https?:\/\//i.test(href)) {
            const a = document.createElement('a');
            a.href = href; a.target = '_blank'; a.rel = 'noopener noreferrer';
            a.style.color = '#38bdf8';
            walk(node, a); dst.appendChild(a);
          } else {
            // showinfo:/killReport:/etc — client-only, render as plain styled text
            const span = document.createElement('span');
            span.className = 'mail-link-disabled';
            walk(node, span); dst.appendChild(span);
          }
          return;
        }

        if (tag === 'FONT') {
          const span = document.createElement('span');
          const rgb = argbToRgb(node.getAttribute('color'));
          if (rgb) span.style.color = rgb;
          walk(node, span); dst.appendChild(span);
          return;
        }

        if (tag === 'LOC') { walk(node, dst); return; }  // unwrap <loc>

        // B / I / U / SPAN / DIV / P — keep the tag, drop all attributes
        const safe = document.createElement(tag.toLowerCase());
        walk(node, safe); dst.appendChild(safe);
      });
    }

    walk(doc.body, out);
    return out;
  }
```
